    <script>
        const API_URL = "https://script.google.com/macros/s/AKfycbxrNnPRL5fcyra9rKJ3Onqkru4krJK0M-YLjAzwPBKPce1DdXgSi7-E044SohztFnWl1A/exec";

        function switchTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(el => el.style.display = 'none');
            document.querySelectorAll('.menu div').forEach(el => el.classList.remove('active'));
            
            if(tabName === 'dashboard') {
                document.getElementById('tab-dashboard').style.display = 'block';
                document.getElementById('header-title').innerText = 'Dashboard Thống Kê Nhân Sự';
                event.currentTarget.classList.add('active');
            } else if(tabName === 'nhansu') {
                document.getElementById('tab-nhansu').style.display = 'block';
                document.getElementById('header-title').innerText = 'Quản Lý Danh Sách Nhân Sự';
                event.currentTarget.classList.add('active');
            }
        }

        async function fetchSystemData() {
            try {
                let response = await fetch(API_URL + "?action=getData");
                let data = await response.json();
                
                // Thống kê Dashboard
                let total = data.nhanSu.length;
                let working = data.nhanSu.filter(i => i.TrangThai === 'Đang làm việc').length;
                let probation = data.nhanSu.filter(i => i.TinhChat === 'Thử việc' || i.TinhChat === 'Học việc').length;
                let departments = data.phongBan.length;

                document.getElementById('stat-total').innerText = total;
                document.getElementById('stat-working').innerText = working;
                document.getElementById('stat-probation').innerText = probation;
                document.getElementById('stat-department').innerText = departments;

                // Đổ dữ liệu vào bảng nhân sự
                let tbody = document.getElementById('employee-table-body');
                tbody.innerHTML = '';
                if(total === 0) {
                    tbody.innerHTML = `<tr><td colspan="7" style="text-align: center;">Chưa có dữ liệu nhân sự</td></tr>`;
                    return;
                }

                data.nhanSu.forEach(emp => {
                    let tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td>${emp.MaNV || ''}</td>
                        <td><b>${emp.HoVaTen || ''}</b></td>
                        <td>${emp.GioiTinh || ''}</td>
                        <td>${emp.ViTri || ''}</td>
                        <td>${emp.DonVi || ''}</td>
                        <td>${emp.SDT || ''}</td>
                        <td><span style="color: ${emp.TrangThai === 'Đang làm việc' ? 'green' : 'red'}">${emp.TrangThai || ''}</span></td>
                    `;
                    tbody.appendChild(tr);
                });

            } catch(e) {
                console.error("Lỗi kết nối dữ liệu:", e);
            }
        }

        // Tự động tải dữ liệu khi mở trang
        fetchSystemData();
    </script>
